// Initializing model-related constants
const int N = 9;
const int xstart = 0;
const int ystart = 0;
const int xtarget = 9;
const int ytarget = 9;
const double p = 0.025;
const double e_gps = 0.2;
formula hasCrashed = (1=0) | (x=0 & y=1) | (x=0 & y=3) ...

// Faulty movement model, also check for crashes
module Robot 
  x : [0..N] init xstart;
  y : [0..N] init ystart;
  crashed : [0..1] init 0;

[west] true -> 
     p: (x'=min(x+1, N)) + 
    p: (y'=min(y+1, N)) + 
    p: (y'=max(y-1, 0)) + 
    (1-3*p): (x'=max(x-1, 0));

[east] true -> 
     (1-3*p): (x'=min(x+1, N)) + 
    p: (y'=min(y+1, N)) + 
    p: (y'=max(y-1, 0)) + 
    p: (x'=max(x-1, 0));

[south] true -> 
     p: (x'=min(x+1, N)) + 
    p: (y'=min(y+1, N)) + 
    (1-3*p): (y'=max(y-1, 0)) + 
    p: (x'=max(x-1, 0));

[north] true -> 
     p: (x'=min(x+1, N)) + 
    (1-3*p): (y'=min(y+1, N)) + 
    p: (y'=max(y-1, 0)) + 
    p: (x'=max(x-1, 0));


  [update_gps] hasCrashed  -> (crashed'=1);
  [update_gps] !hasCrashed -> (crashed'=crashed);
  [skip_update] hasCrashed  -> (crashed'=1);
  [skip_update] !hasCrashed -> (crashed'=crashed);
endmodule

// UAC, action choice based on state estimation xhat, yhat
module Adaptation_MAPE_controller
  [east] (xhat=0) & (yhat=0) -> true;
  [north] (xhat=9) & (yhat=3) -> true;
  
  ...

  [west] (xhat=9) & (yhat=5) -> true;
  [north] (xhat=9) & (yhat=8) -> true;
endmodule

// Knowledge module with faulty gps service
formula due_gps = (step_gps>=c_gps);
module Knowledge
  // Deterministic Knowledge update
  xhat : [0..N] init xstart;
  yhat : [0..N] init ystart;
  step_gps : [1..10] init 1;
  [west] true -> (xhat'=max(xhat-1, 0));

  [east] true -> (xhat'=min(xhat+1, N));

  [south] true -> (yhat'=max(yhat-1, 0));

  [north] true -> (yhat'=min(yhat+1, N));


  [update_gps] due_gps & (x=0) & (y=0) -> 
    (1 - (e_gps/4.0)*1): (xhat'=0) & (yhat'=0) & (step_gps'=1) + 
    (e_gps/4.0): (xhat'=1) & (yhat'=0) & (step_gps'=1);
  
  ...

  [update_gps] due_gps & (x=9) & (y=9) -> 
    (1 - (e_gps/4.0)*1): (xhat'=9) & (yhat'=9) & (step_gps'=1) + 
    (e_gps/4.0): (xhat'=9) & (yhat'=8) & (step_gps'=1);

  [skip_update] !(due_gps) ->(step_gps'=min(step_gps+1,10));
endmodule

rewards "cost"

// Initial thresholds per service
evolve int ua_init_thr_gps [1..10];

// Evolvable (ua_s, observation) -> threshold mapping
evolve int ua_thr_gps_s1_x_0_y_0 [1..10];

...

evolve int ua_thr_gps_s5_x_9_y_9 [1..10];

// Evolvable memory update per (ua_s, observation) -> 1=left, 2=stay, 3=right
evolve int ua_move_s1_x_0_y_0 [1..3];

...

evolve int ua_move_s5_x_9_y_9 [1..3];

// Memory module that sets the thresholds and the transitions
module UA
  ua_s : [1..5] init 1;
  c_gps : [1..10] init 10;

  // Init phase: set initial thresholds (deterministic)
  [UA_INIT_gps] (ua_s=1) & xhat=0 & yhat=0 -> (c_gps'=ua_init_thr_gps);

  // Controller step: deterministic Mealy action + deterministic memory update
  [UA_STEP_gps] (ua_s=1) & xhat=0 & yhat=0 & (ua_move_s1_x_0_y_0=1) -> (c_gps'=ua_thr_gps_s1_x_0_y_0) & (ua_s'=5);
  ...
  [UA_STEP_gps] (ua_s=5) & xhat=9 & yhat=9 & (ua_move_s5_x_9_y_9=3) -> (c_gps'=ua_thr_gps_s5_x_9_y_9) & (ua_s'=1);

endmodule

// Turn module for initialization and synchronization
module Turn
  t : [0..3] init 0;

  // Init phase
  [UA_INIT_gps] (t=0) -> (t'=1);

  // Movement phase
  [east] (t=1) -> (t'=2);
  [west] (t=1) -> (t'=2);
  [north] (t=1) -> (t'=2);
  [south] (t=1) -> (t'=2);

  // Controller phase
  [UA_STEP_gps] (t=2) -> (t'=3);

  // Update / skip phase
  [update_gps] (t=3) -> (t'=1);
  [skip_update] (t=3) -> (t'=1);

endmodule
